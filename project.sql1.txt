-- Total Sales
SELECT SUM(amount) FROM orders;

-- Sales by City
SELECT city, SUM(amount)
FROM orders
GROUP BY city;

-- Orders by Status
SELECT status, COUNT(*)
FROM orders
GROUP BY status;

-- Highest Order Amount
SELECT MAX(amount) FROM orders;

-- Delivered Orders
SELECT *
FROM orders
WHERE status = 'Delivered';

-- Top City by Sales
SELECT city, SUM(amount)
FROM orders
GROUP BY city
ORDER BY SUM(amount) DESC
LIMIT 1;
